## 🎨 Code Generation Standards

### TypeScript First

```typescript
'use client';

import React from 'react';
import { cva, type VariantProps } from 'class-variance-authority';
import { cn } from '@/utils/cn';

// Types defined FIRST
interface ComponentProps extends React.HTMLAttributes<HTMLDivElement> {
  /** Description of prop */
  variant?: 'primary' | 'secondary';
  /** Description of prop */
  size?: 'sm' | 'md' | 'lg';
}

/**
 * ComponentName - Clear, descriptive documentation
 *
 * Longer description if needed, including key features
 *
 * @example
 * <Component variant="primary" size="lg">
 *   Content
 * </Component>
 */
export const Component = React.forwardRef<HTMLDivElement, ComponentProps>(
  ({ className, variant, size, ...props }, ref) => (
    <div ref={ref} className={cn('...', className)} {...props} />
  )
);

Component.displayName = 'Component';
```

### Styling Pattern

```typescript
// Use Tailwind CSS 4 with semantic colors
className={cn(
  // Base styles
  'rounded-lg border transition-colors duration-300',
  // Light mode
  'bg-white text-neutral-900 border-neutral-200',
  // Dark mode
  'dark:bg-neutral-900 dark:text-neutral-100 dark:border-neutral-800',
  // Variants
  variant === 'primary' && 'bg-primary-500 text-white dark:bg-primary-600',
  // States
  'hover:shadow-md focus:outline-none focus:ring-2 focus:ring-offset-2',
  'disabled:opacity-50 disabled:cursor-not-allowed',
  // Custom
  className
)}
```

### Hook Pattern

```typescript
export function useCustomHook(initialValue: string) {
  const [value, setValue] = useState(initialValue);
  const isMountedRef = useRef(true);

  useEffect(() => {
    isMountedRef.current = true;
    return () => {
      isMountedRef.current = false;
    };
  }, []);

  const updateValue = useCallback((newValue: string) => {
    if (isMountedRef.current) {
      setValue(newValue);
    }
  }, []);

  return { value, updateValue };
}
```
